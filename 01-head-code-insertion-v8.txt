<script>
document.addEventListener('DOMContentLoaded', function() {
  const jumpNav = document.getElementById('bsc-jump-nav');
  if (!jumpNav) return;

  // Detect height of QuickSilk site top navigation bar if present
  function getSiteHeaderOffset() {
    // Look for fixed/sticky site header elements
    const possibleHeaders = document.querySelectorAll('header, .site-header, #header, .navbar, .quicksilk-header, [class*="header"], [class*="nav-top"], [class*="navbar"]');
    let maxOffset = 0;
    possibleHeaders.forEach(el => {
      if (el === jumpNav || el.contains(jumpNav)) return;
      const style = window.getComputedStyle(el);
      const rect = el.getBoundingClientRect();
      if ((style.position === 'fixed' || style.position === 'sticky') && rect.top <= 15 && rect.height > 20 && rect.height < 150) {
        if (rect.height > maxOffset) {
          maxOffset = rect.height;
        }
      }
    });
    // Fallback default top offset if fixed site header is present (default ~60px, adjust if needed)
    return maxOffset > 0 ? maxOffset : 60;
  }

  // Placeholder element to reserve vertical layout space when jumpNav is moved to body
  const placeholder = document.createElement('div');
  placeholder.id = 'bsc-jump-nav-placeholder';
  placeholder.style.display = 'none';
  
  const originalParent = jumpNav.parentNode;
  const originalNextSibling = jumpNav.nextSibling;
  originalParent.insertBefore(placeholder, jumpNav);

  let initialTop = 0;
  function updateInitialTop() {
    if (!jumpNav.classList.contains('bsc-is-fixed')) {
      const rect = jumpNav.getBoundingClientRect();
      initialTop = rect.top + window.pageYOffset;
    }
  }
  updateInitialTop();
  window.addEventListener('resize', updateInitialTop);

  const navLinks = jumpNav.querySelectorAll('a[href^="#"]');
  const sections = Array.from(navLinks).map(link => {
    const targetId = link.getAttribute('href').slice(1);
    return document.getElementById(targetId);
  }).filter(Boolean);

  // Smooth scroll with height offset (accounting for top nav + jump nav)
  navLinks.forEach(link => {
    link.addEventListener('click', function(e) {
      e.preventDefault();
      const targetId = this.getAttribute('href').slice(1);
      const targetEl = document.getElementById(targetId);
      if (targetEl) {
        const headerOffset = getSiteHeaderOffset();
        const navHeight = jumpNav.offsetHeight || 50;
        const totalOffset = headerOffset + navHeight + 15;
        const targetPos = targetEl.getBoundingClientRect().top + window.pageYOffset - totalOffset;
        window.scrollTo({
          top: targetPos,
          behavior: 'smooth'
        });
      }
    });
  });

  // Dynamic Viewport Pinning with Top Nav Clearance
  function handleScroll() {
    const scrollPos = window.pageYOffset || document.documentElement.scrollTop;
    const headerOffset = getSiteHeaderOffset();

    if (scrollPos >= (initialTop - headerOffset - 5) && initialTop > 0) {
      if (!jumpNav.classList.contains('bsc-is-fixed')) {
        jumpNav.classList.add('bsc-is-fixed');
        placeholder.style.height = jumpNav.offsetHeight + 'px';
        placeholder.style.marginBottom = window.getComputedStyle(jumpNav).marginBottom || '25px';
        placeholder.style.display = 'block';

        // Move jumpNav directly to <body> so position:fixed attaches to the viewport
        document.body.appendChild(jumpNav);

        jumpNav.style.position = 'fixed';
        jumpNav.style.top = headerOffset + 'px'; // Sits directly below top navigation bar
        jumpNav.style.left = '0';
        jumpNav.style.right = '0';
        jumpNav.style.width = '100%';
        jumpNav.style.maxWidth = '1200px';
        jumpNav.style.margin = '0 auto';
        jumpNav.style.boxSizing = 'border-box';
        jumpNav.style.zIndex = '999999';
        jumpNav.style.borderRadius = '0 0 8px 8px';
        jumpNav.style.boxShadow = '0 10px 30px rgba(0,0,0,0.95)';
        jumpNav.style.backgroundColor = 'rgba(26, 26, 26, 0.95)';
        jumpNav.style.backdropFilter = 'blur(12px)';
        jumpNav.style.webkitBackdropFilter = 'blur(12px)';
      } else {
        // Dynamically keep top updated if headerOffset changes
        jumpNav.style.top = headerOffset + 'px';
      }
    } else {
      if (jumpNav.classList.contains('bsc-is-fixed')) {
        jumpNav.classList.remove('bsc-is-fixed');
        placeholder.style.display = 'none';

        // Move jumpNav back to its original position in the content DOM
        if (originalNextSibling && originalNextSibling.parentNode === originalParent) {
          originalParent.insertBefore(jumpNav, originalNextSibling);
        } else {
          originalParent.appendChild(jumpNav);
        }

        jumpNav.style.position = 'relative';
        jumpNav.style.top = 'auto';
        jumpNav.style.left = 'auto';
        jumpNav.style.right = 'auto';
        jumpNav.style.width = '100%';
        jumpNav.style.margin = '0 0 25px 0';
        jumpNav.style.borderRadius = '6px';
        jumpNav.style.boxShadow = 'none';
        jumpNav.style.backgroundColor = '#1a1a1a';
      }
    }

    // ScrollSpy Active Section Highlighting
    const navHeight = jumpNav.offsetHeight || 50;
    const currentCheckPos = scrollPos + headerOffset + navHeight + 100;

    let currentSection = null;
    sections.forEach(sec => {
      const top = sec.offsetTop;
      const height = sec.offsetHeight;
      if (currentCheckPos >= top && currentCheckPos < top + height) {
        currentSection = sec;
      }
    });

    if ((window.innerHeight + scrollPos) >= document.body.offsetHeight - 60) {
      if (sections.length > 0) currentSection = sections[sections.length - 1];
    }

    navLinks.forEach(link => {
      const targetId = link.getAttribute('href').slice(1);
      if (currentSection && currentSection.id === targetId) {
        link.style.backgroundColor = '#ff4d00';
        link.style.color = '#ffffff';
        link.style.fontWeight = '900';
        link.style.borderRadius = '20px';
        link.style.padding = '5px 14px';
        link.style.boxShadow = '0 2px 10px rgba(255, 77, 0, 0.5)';
      } else {
        link.style.backgroundColor = 'transparent';
        link.style.color = '#aaaaaa';
        link.style.fontWeight = '700';
        link.style.borderRadius = '0';
        link.style.padding = '5px 8px';
        link.style.boxShadow = 'none';
      }
    });
  }

  window.addEventListener('scroll', handleScroll, { passive: true });
  setTimeout(updateInitialTop, 300);
  handleScroll();
});
</script>
